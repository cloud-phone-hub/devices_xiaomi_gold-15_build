           %29 = OpINotEqual %bool %28 %int_0     ← BRANCHZ 条件：r4 != 0
                 OpSelectionMerge %31 None        ← 选择构造，merge=%31
                 OpBranchConditional %29 %31 %30  ← 真→%31(跳过)、假→%30(then)

           %31 = OpLabel                          ← merge 块
