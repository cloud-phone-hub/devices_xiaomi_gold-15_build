## 4. 反汇编（spirv-dis）关键行
%float_2 = OpConstant %float 2
%float_1 = OpConstant %float 1
%float_50 = OpConstant %float 50
%float_0_100000001 = OpConstant %float 0.100000001
%float_0_693147182 = OpConstant %float 0.693147182
%10 = OpExtInst %float %8 InverseSqrt %float_2          # FRSQ.f32
%13 = OpFDiv   %float %float_1 %float_50                 # FRCP.f32 = 1/50
%16 = OpExtInst %float %8 FMax %float_0_100000001 %float_0_693147182   # FMAX.f32
%19 = OpCompositeConstruct %v4float %10 %13 %16 %float_1

## 8. 指令语义小结
- FRSQ.f32  -> GLSL.std.450 InverseSqrt（=32）
- FRCP.f32  -> OpFDiv(1.0, x)（=136；GLSL 无 Recip）
- FMAX.f32  -> GLSL.std.450 FMax（=40）
