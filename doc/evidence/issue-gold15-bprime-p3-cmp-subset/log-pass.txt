       %int_n1 = OpConstant %int -1
        %int_1 = OpConstant %int 1
      %float_1 = OpConstant %float 1
      %float_0 = OpConstant %float 0
  %float_0_100000001 = OpConstant %float 0.100000001
           %20 = OpUGreaterThan %bool %17 %19
           %22 = OpSelect %int %20 %int_n1 %int_0      ← m1：真=0xFFFFFFFF
           %23 = OpUGreaterThan %bool %17 %19
           %25 = OpSelect %int %23 %int_1 %int_0       ← i1：真=1
           %27 = OpFOrdEqual %bool %float_2 %float_2
           %28 = OpSelect %int %27 %int_n1 %int_0      ← FCMP eq，m1
           %30 = OpFOrdGreaterThan %bool %float_2 %float_1
           %32 = OpSelect %float %30 %float_1 %float_0 ← f1：真=1.0
           %38 = OpExtInst %float %14 Fma %36 %float_0_100000001 %float_0
# → ICMP 无符号用 OpUGreaterThan、FCMP 用 OpFOrdEqual/OpFOrdGreaterThan；
#   result_type 编码 i1/m1→OpSelect %int、f1→OpSelect %float，全部正确。
