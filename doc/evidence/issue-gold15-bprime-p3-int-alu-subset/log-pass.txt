## 3) spirv-dis 核对（frag，整数 ALU 行）

    --- ialu ---
         %19 = OpIAdd %int %16 %18          <-- IADD.u32 r4 = r10 + r11
         %20 = OpISub %int %18 %18          <-- ISUB.u32 r5 = r11 - r11 (=0)
         %21 = OpISub %int %16 %18          <-- ISUB.u32 r6 = r10 - r11
         %22 = OpIMul %int %16 %18          <-- IMUL.i32 r7 = r10 * r11
    --- ialu_real ---
         %23 = OpIAdd %int %16 %18          <-- IADD.u32 r4 = r8 + r10^
         %24 = OpISub %int %19 %20          <-- ISUB.u32 r6 = r9 - r7
         %25 = OpIMul %int %18 %22          <-- IMUL.i32 r2 = r10 * r2^
